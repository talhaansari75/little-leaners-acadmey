   127	const newActionId = () => {
   128	  if (typeof crypto !== "undefined" && "randomUUID" in crypto) return crypto.randomUUID();
   129	  const bytes = new Uint8Array(16);
   130	  if (typeof crypto !== "undefined" && "getRandomValues" in crypto) crypto.getRandomValues(bytes);
   131	  else for (let i = 0; i < bytes.length; i++) bytes[i] = Math.floor(Math.random() * 256);
   132	  bytes[6] = (bytes[6]! & 0x0f) | 0x40;
   133	  bytes[8] = (bytes[8]! & 0x3f) | 0x80;
   134	  const h = Array.from(bytes, b => b.toString(16).padStart(2, "0")).join("");
   135	  return `${h.slice(0,8)}-${h.slice(8,12)}-${h.slice(12,16)}-${h.slice(16,20)}-${h.slice(20)}`;

   144	function queueServerGameplayAction(data: ServerGameplayAction) {
   145	  const run = async () => {
   146	    let last: unknown;
   147	    for (let attempt = 0; attempt < 3; attempt++) {
   148	      try {
   149	        const result = await recordGameplayAction({ data });
   150	        if (!result.ok) throw new Error(result.error ?? "Gameplay action rejected by server");
   151	        if (result.save) get().applyServerSave(result.save as PlayerSave);
   152	        if (result.hintEconomy) {
   153	          get().patchSave((s) => ({ ...s, coins: result.hintEconomy!.coins, stats: { ...s.stats, hintsUsed: result.hintEconomy!.hintsUsed } }));
   154	        }
   155	        return result;
   156	      } catch (error) { last = error; if (attempt < 2) await sleep(75 * 2 ** attempt); }
   157	    }
   158	    return { ok: false as const, error: last instanceof Error ? last.message : "Gameplay action delivery failed" };
   159	  };
   160	  const previous = serverActionQueues.get(data.sessionId) ?? Promise.resolve();
   161	  const next = previous.catch(() => undefined).then(run);
   162	  serverActionQueues.set(data.sessionId, next);
   163	  void next.then(() => {
   164	    if (serverActionQueues.get(data.sessionId) === next) serverActionQueues.delete(data.sessionId);
   165	  }, () => {
   166	    if (serverActionQueues.get(data.sessionId) === next) serverActionQueues.delete(data.sessionId);
   167	  });
   168	  return next;
   169	}
