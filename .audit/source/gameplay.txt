    40	export const startGameplaySession = createServerFn({ method: "POST" })
    41	  .middleware([authMiddleware])
    42	  .validator((d: { kind: "level" | "daily"; level: number; day?: string; mode: string; language: string; dailyChallengeId?: string }) => ({
    43	    kind: d.kind === "daily" ? "daily" as const : "level" as const,
    44	    level: Math.max(1, Math.min(2000, Math.floor(Number(d.level) || 1))),
    45	    day: String(d.day ?? "").slice(0, 10),
    46	    mode: modes.has(d.mode as GameMode) ? d.mode as GameMode : "classic" as GameMode,
    47	    language: langs.has(d.language as LangCode) ? d.language as LangCode : "en" as LangCode,
    48	    dailyChallengeId: d.dailyChallengeId ? String(d.dailyChallengeId).slice(0, 20) : undefined,
    49	  }))
    50	  .handler(async ({ context, data }) => {
    51	    if (data.kind === "daily" && data.day !== todayKey()) return { ok:false as const, error:"Daily challenge is not for today." };
    52	    if (data.kind === "daily" && data.dailyChallengeId !== dailyChallengeFor(data.day).id) return { ok:false as const, error:"Daily challenge variant mismatch." };
    53	    const db = getPrisma();
    54	    if (data.kind === "daily") {
    55	      const claimed = await db.dailyResult.findUnique({ where: { userId_dayKey: { userId: context.userId, dayKey: data.day } }, select: { userId: true } });
    56	      if (claimed) return { ok:false as const, error:"Today's daily challenge has already been claimed." };
    57	    }
    58	    const row = await db.playerSave.findUnique({ where:{userId:context.userId} });
    59	    const save = row?.saveJson ? safeJson(JSON.parse(row.saveJson)) : safeJson(defaultSave());
    60	    const unlocked = Number(save.unlockedLevel ?? 1);
    61	    if (data.kind === "level" && data.level > unlocked) return { ok:false as const, error:"Level is locked on the server." };
    62	    const rules = modeRules(data.mode);
    63	    const equippedPet = typeof save.equippedPet === "string" ? save.equippedPet : null;
    64	    const equippedPetLevel = Number(save.petLevels?.[equippedPet ?? ""] ?? 1);
    65	    const effects = petEffect(equippedPet, equippedPetLevel);
    66	    const adaptive = data.kind === "level" ? intelligenceAdaptivePlan(save as unknown as import("@/lib/game/types").PlayerSave, data.level) : { tier: "steady" as const, timeMultiplier: 1, startingReveals: 0, bonusTarget: 0, surprise: false };
    67	    const energyCost = Math.max(1, Math.ceil(1 * rules.energyMultiplier * (1 - Number(effects.energyReductionPercent ?? 0) / 100)));
    68	    const energy = Number(save.energy ?? 20);
    69	    if (data.kind === "level" && !rules.free && energy < energyCost) return { ok:false as const, error:"Not enough energy." };
    70	    const puzzle = data.kind === "daily" ? puzzleForDaily(data.day, data.language, data.dailyChallengeId) : puzzleForLevel(data.level, data.mode, data.language);
    71	    const daily = data.kind === "daily" ? dailyChallengeFor(data.day) : undefined;
    72	    const spec = data.kind === "daily" ? { timeLimit: daily?.id === "speed" ? 90 : 150 } : modeMods(data.mode, specFor(data.level));
    73	    const timeLimitMs = Math.max(20_000, Math.floor((spec.timeLimit ?? 10 * 60) * 1000 * (1 + Number(effects.timeBonusPercent ?? 0) / 100) * adaptive.timeMultiplier));
    74	    const sessionId = randomUUID();
    75	    const now = Date.now();
    76	    const state: SessionState = { kind:data.kind, level:data.level, day:data.day, mode:data.mode, language:data.language, dailyChallengeId:data.dailyChallengeId, words:puzzle.words.map(w=>w.toUpperCase()), grid:puzzle.grid, placements:puzzle.placements.map(p=>({word:p.word.toUpperCase(),cells:p.cells})), actions:[], timeLimitMs, adaptiveTier: adaptive.tier, adaptiveBonusTarget: adaptive.bonusTarget, startingReveals: adaptive.startingReveals };
    77	    await db.$transaction(async tx => {
    78	      if (data.kind === "daily") {
    79	        await tx.$executeRaw`SELECT pg_advisory_xact_lock(hashtext(${`daily-session:${context.userId}:${data.day}`}))`;
    80	        const claimedInsideTx = await tx.dailyResult.findUnique({ where: { userId_dayKey: { userId: context.userId, dayKey: data.day } }, select: { userId: true } });
    81	        if (claimedInsideTx) throw new Error("Today's daily challenge has already been claimed.");
    82	        const activeDaily = await tx.gameSessionV5.findMany({ where:{userId:context.userId,levelId:0,status:"open"}, select:{stateJson:true}, take:10 });
    83	        if (activeDaily.some((session) => String((safeJson(session.stateJson)).day ?? "") === data.day)) {
    84	          throw new Error("Today's daily challenge is already active.");
    85	        }
    86	      }
    87	      const currentRow = await tx.playerSave.findUnique({ where:{userId:context.userId} });
    88	      const currentSave = currentRow?.saveJson ? safeJson(JSON.parse(currentRow.saveJson)) : safeJson(defaultSave());
    89	      const currentUnlocked = Number(currentSave.unlockedLevel ?? 1);
    90	      const energyAt = Number(currentSave.energyAt ?? now);
    91	      const refills = Math.max(0, Math.floor((now - energyAt) / ENERGY_REFILL_MS));
    92	      const currentEnergy = Math.min(MAX_ENERGY, Number(currentSave.energy ?? MAX_ENERGY) + refills);
    93	      const nextEnergyAt = currentEnergy >= MAX_ENERGY ? now : energyAt + refills * ENERGY_REFILL_MS;
    94	      if (data.kind === "level" && data.level > currentUnlocked) throw new Error("Level is locked on the server.");
    95	      if (data.kind === "level" && !rules.free && currentEnergy < energyCost) throw new Error("Not enough energy.");
    96	      if (data.kind === "level" && !rules.free) {
    97	        const nextSave = {...currentSave, energy:currentEnergy-energyCost, energyAt:nextEnergyAt};
    98	        if (currentRow) await tx.playerSave.update({ where:{userId:context.userId}, data:{ saveJson:JSON.stringify(nextSave), version:{increment:1}, revision:{increment:1n} } });
    99	        else await tx.playerSave.create({ data:{userId:context.userId,saveJson:JSON.stringify(nextSave),version:1,revision:1n} });
   100	      } else if (!currentRow) {
   101	        await tx.playerSave.create({ data:{userId:context.userId,saveJson:JSON.stringify({...currentSave,energy:currentEnergy,energyAt:nextEnergyAt}),version:1,revision:1n} });
   102	      } else if (currentEnergy !== Number(currentSave.energy ?? MAX_ENERGY) || nextEnergyAt !== energyAt) {
   103	        await tx.playerSave.update({ where:{userId:context.userId}, data:{saveJson:JSON.stringify({...currentSave,energy:currentEnergy,energyAt:nextEnergyAt}),version:{increment:1},revision:{increment:1n}} });
   104	      }
   105	      await tx.gameSessionV5.create({ data:{id:sessionId,userId:context.userId,levelId:data.level,seed:String(puzzle.seed),status:"open",startedAt:BigInt(now),updatedAt:BigInt(now),score:0,stateJson:state as any} });
   106	    }, { isolationLevel: "Serializable" });
   107	    const authoritativeRow = await db.playerSave.findUnique({ where:{userId:context.userId}, select:{saveJson:true} });
   108	    const authoritativeSave = authoritativeRow?.saveJson ? JSON.parse(authoritativeRow.saveJson) : defaultSave();
   109	    return { ok:true as const, sessionId, startedAt:now, timeLimitMs, seed:puzzle.seed, adaptiveTier: adaptive.tier, startingReveals: adaptive.startingReveals, save: authoritativeSave };
   110	  });

   112	export const recordGameplayAction = createServerFn({ method:"POST" })
   113	  .middleware([authMiddleware])
   114	  .validator((d:{sessionId:string; actionId?:string; type:string; word?:string; cells?:Cell[]; hintKind?:string})=>{
   115	    const allowed = new Set(["found","bonus","miss","hint","pause","resume"]);
   116	    if (!allowed.has(d.type)) throw new Error("Invalid gameplay action type.");
   117	    if (d.type === "hint" && d.hintKind !== "first" && d.hintKind !== "letter" && d.hintKind !== "word") throw new Error("Invalid hint kind.");
   118	    return {
   119	      sessionId:String(d.sessionId??"").trim().slice(0,64),
   120	      actionId:String(d.actionId??"").trim(),
   121	      type:d.type as Action["type"],
   122	      word:d.word ? String(d.word).toUpperCase().slice(0,32):undefined,
   123	      cells:Array.isArray(d.cells)?d.cells.slice(0,32).map(c=>[Number(c[0]),Number(c[1])] as Cell):undefined,
   124	      hintKind:d.hintKind === "first" || d.hintKind === "letter" || d.hintKind === "word" ? d.hintKind : undefined,
   125	    };
   126	  })
   127	  .handler(async({context,data})=>{
   128	    if (!/^.{8,64}$/.test(data.sessionId) || !/^[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(data.actionId)) {
   129	      return {ok:false as const,error:"Invalid gameplay action identity."};
   130	    }
   131	    const db=getPrisma();
   132	    return db.$transaction(async tx=>{
   133	      await tx.$queryRaw`SELECT id FROM game_sessions_v5 WHERE id=${data.sessionId} AND user_id=${context.userId} FOR UPDATE`;
   134	      const row=await tx.gameSessionV5.findUnique({where:{id:data.sessionId}});
   135	      if(!row || row.userId!==context.userId || row.status!=="open") return {ok:false as const,error:"Gameplay session is not active."};
   136	      const state=JSON.parse(JSON.stringify(row.stateJson)) as SessionState;
   137	      if (data.actionId) {
   138	        const hasDuplicateAction = state.actions.some(a => a.id === data.actionId);
   139	        const duplicateAction = hasDuplicateAction ? state.actions.find(a => a.id === data.actionId) : undefined;
   140	        if (duplicateAction) {
   141	          if (duplicateAction.type === "hint") {
   142	            const hintSaveRow = await tx.playerSave.findUnique({where:{userId:context.userId},select:{saveJson:true}});
   143	            const hintSave = safeJson(hintSaveRow?.saveJson ? JSON.parse(hintSaveRow.saveJson) : defaultSave());
   144	            return {ok:true as const,duplicate:true,hintEconomy:{coins:Number(hintSave.coins ?? 0),hintsUsed:Number(safeJson(hintSave.stats).hintsUsed ?? 0)}};
   145	          }
   146	          return {ok:true as const,duplicate:true};
   147	        }
   148	      }
   149	      if(state.actions.length>=500) return {ok:false as const,error:"Too many gameplay actions."};
   150	      const lastAction = state.actions[state.actions.length - 1];
   151	      const currentlyPaused = lastAction?.type === "pause";
   152	      if(data.type === "pause" && currentlyPaused) return {ok:false as const,error:"Gameplay session is already paused."};
   153	      if(data.type === "resume" && !currentlyPaused) return {ok:false as const,error:"Gameplay session is not paused."};
   154	      if(data.type === "pause" && state.kind === "daily" && state.dailyChallengeId === "speed") return {ok:false as const,error:"Speed Day cannot be paused."};
   155	      let specialKinds: string[] | undefined;
   156	      const puzzle = state.kind === "daily" ? puzzleForDaily(state.day, state.language, state.dailyChallengeId) : puzzleForLevel(state.level, state.mode, state.language);
   157	      const specialMap = new Map(specialTilesForPuzzle(puzzle).map(tile => [`${tile.cell[0]},${tile.cell[1]}`, tile.kind] as const));
   158	      if(data.type === "found") {
   159	        const p=state.placements.find(x=>x.word===data.word);
   160	        if(!p || !data.cells || !sameCells(data.cells,p.cells)) return {ok:false as const,error:"Invalid word path."};
   161	        if(state.actions.some(a=>a.type==="found" && a.word===data.word)) return {ok:true as const,duplicate:true};
   162	        specialKinds = [...specialKindsAt(data.cells, specialMap)];
   163	      }
   164	      if ((data.type === "found" || data.type === "bonus") && data.cells) {
   165	        const rows = state.grid.length;
   166	        const cols = rows > 0 ? state.grid[0]!.length : 0;
   167	        if (data.cells.some(([r,c]) => !Number.isInteger(r) || !Number.isInteger(c) || r < 0 || c < 0 || r >= rows || c >= cols)) {
   168	          return {ok:false as const,error:"Gameplay path contains an out-of-bounds cell."};
   169	        }
   170	      }
   171	      if(data.type === "bonus") {
   172	        if(!data.word || !data.cells || data.cells.length < 2 || data.cells.length !== data.word.length) return {ok:false as const,error:"Invalid bonus word."};
   173	        const bonusKey = bonusActionKey(data.word, data.cells);
   174	        if (state.actions.some(a => a.type === "bonus" && bonusActionKey(a.word ?? "", a.cells ?? []) === bonusKey)) return {ok:true as const,duplicate:true};
   175	        const letters = data.cells.map(([r,c]) => state.grid[r]?.[c] ?? "").join("").toUpperCase();
   176	        const reverse = [...letters].reverse().join("");
   177	        const word = data.word.toUpperCase();
   178	        if (letters !== word && reverse !== word) return {ok:false as const,error:"Bonus path does not match the grid."};
   179	        if (!isWord(word)) return {ok:false as const,error:"Bonus word is not in the verified word bank."};
   180	        specialKinds = [...specialKindsAt(data.cells, specialMap)];
   181	        for (let i=1;i<data.cells.length;i++) {
   182	          const [r1,c1]=data.cells[i-1]!, [r2,c2]=data.cells[i]!;
   183	          if (Math.abs(r2-r1)>1 || Math.abs(c2-c1)>1 || (r1===r2 && c1===c2)) return {ok:false as const,error:"Bonus path is not contiguous."};
   184	        }
   185	      }
   186	      if ((data.type === "found" || data.type === "bonus") && data.cells) {
   187	        const clipped = clipAtLocked(data.cells, specialMap);
   188	        if (clipped.length !== data.cells.length) return {ok:false as const,error:"Locked tile blocks this path."};
   189	        // Moving hazards are server-evaluated against a deterministic session tick.
   190	        const tick = Math.floor((Date.now() - Number(row.startedAt)) / 1400);
   191	        const movingTiles = specialTilesForPuzzle(puzzle).filter(t => t.kind === "moving");
   192	        if (movingTiles.length) {
   193	          const movingMap = new Map(movingTiles.map((t,i) => [`${t.cell[0]},${t.cell[1]}`, i]));
   194	          if (data.cells.some(c => movingMap.has(`${c[0]},${c[1]}`) && ((tick + (movingMap.get(`${c[0]},${c[1]}`) ?? 0)) % movingTiles.length === 0))) return {ok:false as const,error:"A moving obstacle blocked that path."};
   195	        }
   196	      }
   197	      let hintEconomy: { coins: number; hintsUsed: number } | undefined;
   198	      if (data.type === "hint") {
   199	        const hintKind = data.hintKind!;
   200	        // Serialize economy mutations across every gameplay session for this user.
   201	        await tx.$executeRaw`SELECT pg_advisory_xact_lock(hashtext(${`hint-economy:${context.userId}`}))`;
   202	        const saveRow = await tx.playerSave.findUnique({where:{userId:context.userId}});
   203	        const save = safeJson(saveRow?.saveJson ? JSON.parse(saveRow.saveJson) : defaultSave());
   204	        const equippedPet = typeof save.equippedPet === "string" ? save.equippedPet : null;
   205	        const petLevel = Number(save.petLevels?.[equippedPet ?? ""] ?? 1);
   206	        if (modeRules(state.mode).noHints) return {ok:false as const,error:"Hints are disabled in this mode."};
   207	        const cost = Math.max(5, HINT_COST[hintKind] - (equippedPet === "fox" && hintKind !== "word" ? 8 : 0));
   208	        const coins = Number(save.coins ?? 0);
   209	        if (coins < cost) return {ok:false as const,error:"Not enough coins for that hint."};
   210	        const stats = safeJson(save.stats);
   211	        const nextSave = {...save,coins:coins-cost,stats:{...stats,hintsUsed:Number(stats.hintsUsed ?? 0)+1}};
   212	        if (saveRow) await tx.playerSave.update({where:{userId:context.userId},data:{saveJson:JSON.stringify(nextSave),version:{increment:1},revision:{increment:1n}}});
   213	        else await tx.playerSave.create({data:{userId:context.userId,saveJson:JSON.stringify(nextSave),version:1,revision:1n}});
   214	        hintEconomy = { coins: coins - cost, hintsUsed: Number((nextSave.stats as Record<string, unknown>).hintsUsed ?? 0) };
   215	      }
   216	      state.actions.push({id:data.actionId || undefined,type:data.type,word:data.word,cells:data.cells,specialKinds,at:Date.now(),hintKind:data.hintKind});
   217	      await tx.gameSessionV5.update({where:{id:data.sessionId},data:{stateJson:state as any,updatedAt:BigInt(Date.now())}});
   218	      return {ok:true as const, ...(hintEconomy ? {hintEconomy} : {})};
   219	    }, { isolationLevel: "Serializable" });
   220	  });

   225	  .validator((d: { level: number }) => ({ level: Math.max(1, Math.min(2000, Math.floor(Number(d.level) || 1))) }))
   226	  .handler(async ({ context, data }) => {

   228	    const db = getPrisma();
   229	    const permanentRewardKey = `boss-reward:${context.userId}:${data.level}`;

   231	    if (alreadyDefeated) return { ok:false as const, error:"This World Boss has already been defeated." };
   232	    const row = await db.playerSave.findUnique({ where:{userId:context.userId} });
   233	    const save = safeJson(row?.saveJson ? JSON.parse(row.saveJson) : defaultSave());

   238	      if (existing) return { ok:true as const, sessionId:existing.id, combat:existing.stateJson as CombatState };
   239	    const gear = Array.isArray(save.equipment) ? save.equipment.reduce((n:any,e:any) => n + ((save.equippedEquipment?.[e.slot] === e.id) ? Number(e.power||0) : 0),0) : 0;
   240	    const petId = typeof save.equippedPet === "string" ? save.equippedPet : null;
   241	    const petLevel = Number(save.petLevels?.[petId ?? ""] ?? 1);
   242	    const power = playerCombatPower(gear, Number(save.skills?.speed ?? 0), petPower(petId, petLevel));

   244	    const id = randomUUID(), now=Date.now();
   245	      await tx.gameSessionV5.create({data:{id,userId:context.userId,levelId:data.level,seed:`boss:${data.level}`,status:"boss_open",startedAt:BigInt(now),updatedAt:BigInt(now),score:0,stateJson:{...combat,playerPower:power,petAbility:petProfile(petId)?.ability ?? null} as any}});
   246	      return {ok:true as const,sessionId:id,combat};
   247	    }, { isolationLevel:"Serializable" });
   248	  });

   251	  .middleware([authMiddleware])
   252	  .validator((d:{sessionId:string;action:string;perfect?:boolean})=>({sessionId:String(d.sessionId??"").slice(0,64),action:(d.action==="word"||d.action==="guard"||d.action==="power")?d.action:"word",perfect:Boolean(d.perfect)}))

   259	      const state=safeJson(row.stateJson) as any as CombatState & {playerPower:number;petAbility?:string|null;rewardClaimed?:boolean};
   260	      if(state.victory || state.defeated) return {ok:true as const,combat:state};
