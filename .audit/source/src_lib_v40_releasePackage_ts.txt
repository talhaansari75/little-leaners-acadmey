    23	function fnv1a(input:string) { let h=2166136261; for(let i=0;i<input.length;i++) h=Math.imul(h ^ input.charCodeAt(i),16777619); return (h>>>0).toString(16).padStart(8,"0"); }

    25	export async function digestFor(input:string):Promise<{algorithm:"SHA-256"|"FNV-1a";digest:string}> {
    26	  try {
    27	    if (globalThis.crypto?.subtle) {
    28	      const bytes=new TextEncoder().encode(input);
    29	      const hash=await crypto.subtle.digest("SHA-256",bytes);
    30	      return {algorithm:"SHA-256",digest:Array.from(new Uint8Array(hash)).map(x=>x.toString(16).padStart(2,"0")).join("")};
    31	    }
    32	  } catch { /* deterministic fallback below */ }
    33	  return {algorithm:"FNV-1a",digest:fnv1a(input)};
    34	}

    36	export async function buildReleasePackage(draftId:string):Promise<CreatorReleasePackage|null>{
    37	  const draft=listDrafts().find(d=>d.id===draftId); if(!draft)return null;
    38	  const readiness=evaluatePublishReadiness(draftId); if(!readiness)return null;
    39	  const data=canonicalReleaseData(draft,readiness);
    40	  return {schema:"mera-word-search-journey.creator-release",version:1,exportedAt:new Date().toISOString(),appVersion:"v40",draft,
    41	    readiness:{score:readiness.score,ready:readiness.ready,blockers:readiness.blockers,recommendations:readiness.recommendations,auditScore:readiness.audit.score,latestPlaytestScore:readiness.latest?.score??null},
    42	    integrity:await digestFor(data)};
    43	}
