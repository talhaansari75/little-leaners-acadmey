   118	export function emptyLearningProfile(): LearningProfile {
   119	  const byClass = {} as Record<LearningClass, Record<string, SkillStat>>;
   120	
   121	  for (const name of LEARNING_CLASSES) {
   122	    byClass[name] = {};
   123	  }
   124	
   125	  return {
   126	    byClass,
   127	    lastClass: "Nursery",
   128	    updatedAt: 0,
   129	  };
   130	}

   132	export function normalizeProfile(input: unknown): LearningProfile {
   133	  const fallback = emptyLearningProfile();
   134	
   135	  if (!input || typeof input !== "object") {
   136	    return fallback;
   137	  }
   138	
   139	  const source = input as Partial<LearningProfile>;
   140	
   141	  for (const className of LEARNING_CLASSES) {
   142	    const sourceStats =
   143	      source.byClass &&
   144	      typeof source.byClass === "object" &&
   145	      source.byClass[className] &&
   146	      typeof source.byClass[className] === "object"
   147	        ? source.byClass[className]
   148	        : {};
   149	
   150	    fallback.byClass[className] = {};
   151	
   152	    for (const skill of CLASS_SKILLS[className]) {
   153	      const raw = (sourceStats as Record<string, unknown>)[skill];
   154	
   155	      if (raw && typeof raw === "object") {
   156	        const stat = raw as Partial<SkillStat>;
   157	
   158	        fallback.byClass[className][skill] = {
   159	          attempts: Number(stat.attempts) || 0,
   160	          correct: Number(stat.correct) || 0,
   161	          totalTimeMs: Number(stat.totalTimeMs) || 0,
   162	          lastAttemptAt: Number(stat.lastAttemptAt) || 0,
   163	          signal:
   164	            stat.signal === "game" ||
   165	            stat.signal === "picture" ||
   166	            stat.signal === "audio" ||
   167	            stat.signal === "story" ||
   168	            stat.signal === "hands-on"
   169	              ? stat.signal
   170	              : undefined,
   171	        };
   172	      }
   173	    }
   174	  }
   175	
   176	  if (
   177	    typeof source.lastClass === "string" &&
   178	    LEARNING_CLASSES.includes(source.lastClass as LearningClass)
   179	  ) {
   180	    fallback.lastClass = source.lastClass as LearningClass;
   181	  }
   182	
   183	  fallback.updatedAt = Number(source.updatedAt) || 0;
   184	
   185	  return fallback;
   186	}

   246	export function recommendNext(
   247	  profile: LearningProfile,
   248	  className: LearningClass = profile.lastClass,
   249	  now = Date.now(),
   250	): Recommendation {
   251	  const normalized = normalizeProfile(profile);
   252	  const skills = CLASS_SKILLS[className];
   253	  const stats = normalized.skills;
   254	
   255	  let bestSkill = skills[0];
   256	  let lowestScore = Number.POSITIVE_INFINITY;
   257	
   258	  for (const skill of skills) {
   259	    const stat = stats[skill];
   260	
   261	    if (!stat || stat.attempts === 0) {
   262	      bestSkill = skill;
   263	      break;
   264	    }
   265	
   266	    const accuracy = stat.correct / Math.max(1, stat.attempts);
   267	    const recentPenalty =
   268	      stat.lastAttemptAt > 0 &&
   269	      now - stat.lastAttemptAt < 24 * 60 * 60 * 1000
   270	        ? 0.1
   271	        : 0;
   272	
   273	    const score = accuracy - recentPenalty;
   274	
   275	    if (score < lowestScore) {
   276	      lowestScore = score;
   277	      bestSkill = skill;
   278	    }
   279	  }
   280	
   281	  const stat = stats[bestSkill];
   282	  const accuracy = stat?.attempts
   283	    ? stat.correct / stat.attempts
   284	    : 0;
   285	
   286	  const difficulty =
   287	    !stat || stat.attempts < 2
   288	      ? "gentle"
   289	      : accuracy >= 0.85
   290	        ? "challenge"
   291	        : accuracy < 0.6
   292	          ? "gentle"
   293	          : "steady";
   294	
   295	  const mode: LearningSignal =
   296	    stat?.signal ??
   297	    (className === "Montessori"
   298	      ? "picture"
   299	      : className === "Nursery"
   300	        ? "game"
   301	        : "hands-on");
   302	
   303	  return {
   304	    className,
   305	    skill: bestSkill,
   306	    difficulty,
   307	    mode,
   308	    title: `${bestSkill.replaceAll("-", " ")} practice`,
   309	    reason:
   310	      !stat || stat.attempts === 0
   311	        ? "This is a fresh skill for this class."
   312	        : accuracy < 0.6
   313	          ? "A little more practice will help this skill grow."
   314	          : accuracy >= 0.85
   315	            ? "This skill is going well, so a small challenge is ready."
   316	            : "Keep building this skill with another practice round.",
   317	    confidence: stat?.attempts
   318	      ? Math.min(99, 60 + Math.min(35, stat.attempts * 5))
   319	      : 60,
   320	  };
   321	}

   323	const PROFILE_KEY = "lla-learning-profile";
   324	
   325	export function getLearningProfile(): LearningProfile {
   326	  if (typeof window === "undefined") {
   327	    return emptyLearningProfile();
   328	  }
   329	
   330	  try {
   331	    const raw = window.localStorage.getItem(PROFILE_KEY);
   332	    return raw ? normalizeProfile(JSON.parse(raw)) : emptyLearningProfile();
   333	  } catch {
   334	    return emptyLearningProfile();
   335	  }
   336	}
   337	
   338	function saveLearningProfile(profile: LearningProfile) {
   339	  if (typeof window === "undefined") return;
   340	
   341	  try {
   342	    window.localStorage.setItem(
   343	      PROFILE_KEY,
   344	      JSON.stringify(profile),
   345	    );
   346	  } catch {
   347	    // Ignore storage failures.
   348	  }
   349	}

   351	export function recordAttempt(
   352	  profile: LearningProfile,
   353	  skill: string,
   354	  correct: boolean,
   355	  signal?: LearningSignal,
   356	  now = Date.now(),
   357	): LearningProfile {
   358	  const current = profile.skills[skill] ?? {
   359	    attempts: 0,
   360	    correct: 0,
   361	    lastAttemptAt: 0,
   362	    signal,
   363	  };
   364	
   365	  return {
   366	    ...profile,
   367	    skills: {
   368	      ...profile.skills,
   369	      [skill]: {
   370	        ...current,
   371	        attempts: current.attempts + 1,
   372	        correct: current.correct + (correct ? 1 : 0),
   373	        lastAttemptAt: now,
   374	        ...(signal ? { signal } : {}),
   375	      },
   376	    },
   377	  };
   378	}
   379	
   380	export function recordLearningSignal(
   381	  profile: LearningProfile,
   382	  skill: string,
   383	  signal: LearningSignal,
   384	  correct = false,
   385	  now = Date.now(),
   386	): LearningProfile {
   387	  return recordAttempt(profile, skill, correct, signal, now);
