     1	import { auditPuzzleWordsLocal, type PuzzleAudit } from "@/lib/v36_puzzleAudit";
     2	import { listPlaytestResults, type PlaytestResult } from "@/lib/v38/playtest/playtest";
     3	import { listDrafts, type CreatorDraft } from "@/lib/v24/creator/creatorService";
     4	
     5	export type PublishReadiness = {
     6	  draft: CreatorDraft;
     7	  audit: PuzzleAudit;
     8	  playtests: PlaytestResult[];
     9	  latest?: PlaytestResult;
    10	  score: number;
    11	  ready: boolean;
    12	  blockers: string[];
    13	  recommendations: string[];
    14	};
    15	
    16	export function evaluatePublishReadiness(draftId: string): PublishReadiness | null {
    17	  const draft = listDrafts().find(d => d.id === draftId);
    18	  if (!draft) return null;
    19	  const audit = auditPuzzleWordsLocal(draft.words);
    20	  const playtests = listPlaytestResults().filter(r => r.puzzleId === draftId);
    21	  const latest = playtests[0];
    22	  const blockers: string[] = [];
    23	  const recommendations: string[] = [];
    24	  if (!audit.ok) blockers.push("Puzzle QA audit is below the publish threshold.");
    25	  if (!latest?.completed) blockers.push("Complete at least one full creator playtest.");
    26	  if (latest && latest.score < 60) blockers.push("Latest playtest score is below 60.");
    27	  if (draft.words.length < 5) recommendations.push("Add more target words for a richer puzzle.");
    28	  if (audit.score < 80) recommendations.push("Improve word-length variety and vocabulary quality.");
    29	  if (latest && latest.mistakes > Math.max(2, Math.ceil(draft.words.length * 0.25))) recommendations.push("Retest after checking ambiguous or hard-to-find placements.");
    30	  const score = Math.round((Math.min(100, audit.score) * 0.6) + (latest ? latest.score * 0.4 : 0));
    31	  return { draft, audit, playtests, latest, score, ready: blockers.length === 0, blockers, recommendations };
    32	}
    33	
    34	export function listPublishCandidates(): PublishReadiness[] {
    35	  return listDrafts().filter(d => d.status !== "archived").map(d => evaluatePublishReadiness(d.id)).filter((x): x is PublishReadiness => Boolean(x));
    36	}
