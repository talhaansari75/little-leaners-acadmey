     8	const PRODUCTS = {
     9	  ad_free: { name: "Ad-Free Journey", priceEnv: "STRIPE_PRICE_AD_FREE", mode: "payment" },
    10	  premium: { name: "Journey Premium", priceEnv: "STRIPE_PRICE_PREMIUM", mode: "subscription" },
    11	  starter_pack: { name: "Starter Pack", priceEnv: "STRIPE_PRICE_STARTER", mode: "payment" },
    12	} as const;
    13	
    14	export const createCheckoutSession = createServerFn({method:"POST"})
    15	  .middleware([authMiddleware])
    16	  .validator((d:{productId:string})=>({productId:String(d.productId??"").slice(0,64)}))
    17	  .handler(async({context,data})=>{
    18	    const product = PRODUCTS[data.productId as keyof typeof PRODUCTS];
    19	    const secret = process.env.STRIPE_SECRET_KEY?.trim();
    20	    const priceId = product ? process.env[product.priceEnv]?.trim() : undefined;
    21	    if (!product || !secret || !priceId) return {ok:false as const,error:"Payment provider is not configured for this product."};
    22	    const request = await import("@tanstack/react-start/server").then((m)=>m.getRequest());
    23	    const origin = request?.headers.get("origin") || process.env.APP_URL?.trim() || "http://localhost:8080";
    24	    const params = new URLSearchParams();
    25	    params.set("mode", product.mode);
    26	    params.set("line_items[0][price]", priceId);
    27	    params.set("line_items[0][quantity]", "1");
    28	    params.set("success_url", `${origin}/?payment=success`);
    29	    params.set("cancel_url", `${origin}/?payment=cancelled`);
    30	    params.set("client_reference_id", context.userId);
    31	    params.set("metadata[userId]", context.userId);
    32	    params.set("metadata[productId]", data.productId);
    33	    if (product.mode === "payment") {
    34	      params.set("payment_intent_data[metadata][userId]", context.userId);
    35	      params.set("payment_intent_data[metadata][productId]", data.productId);
    36	    }
    37	    if (product.mode === "subscription") {
    38	      params.set("subscription_data[metadata][userId]", context.userId);
    39	      params.set("subscription_data[metadata][productId]", data.productId);
    40	    }
    41	    const response = await fetch("https://api.stripe.com/v1/checkout/sessions", {
    42	      method:"POST",
    43	      headers:{Authorization:`Bearer ${secret}`,"Content-Type":"application/x-www-form-urlencoded"},
    44	      body:params.toString(),
    45	    });
    46	    const body = await response.json().catch(()=>null) as {url?:string;error?:{message?:string}} | null;
    47	    if (!response.ok || !body?.url) return {ok:false as const,error:body?.error?.message || "Unable to create checkout session."};
    48	    return {ok:true as const,url:body.url};
    49	  });
