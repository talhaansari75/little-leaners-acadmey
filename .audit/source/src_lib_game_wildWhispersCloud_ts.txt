     1	import { createServerFn } from "@tanstack/react-start";
     2	import { authMiddleware } from "@/lib/auth/middleware";
     3	import { getPrisma } from "@/lib/db";
     4	import { defaultSave } from "./persist";
     5	import { WILD_ANIMALS, type WildDifficulty, type WildMode, type WildProgress } from "./wildWhispers";
     6	
     7	const modes = ["normal", "silhouette", "sound"] as const;
     8	const difficulties = ["easy", "medium", "hard"] as const;
     9	
    10	type StoredSave = Record<string, unknown> & { wildWhispers?: WildProgress; wildWhispersChallenge?: { animalId: string; issuedAt: number } };
    11	
    12	const fallback: WildProgress = { discovered: [], bestStreak: 0, streak: 0, gamesPlayed: 0, score: 0, lives: 3, difficulty: "medium", mode: "normal" };
    13	
    14	const normalize = (value: unknown): WildProgress => {
    15	  const v = (value && typeof value === "object" ? value : {}) as Partial<WildProgress>;
    16	  return {
    17	    discovered: Array.isArray(v.discovered) ? [...new Set(v.discovered.map(String).filter((id) => WILD_ANIMALS.some((a) => a.id === id)).slice(0, WILD_ANIMALS.length))] : [],
    18	    bestStreak: Math.max(0, Math.min(10000, Math.floor(Number(v.bestStreak) || 0))),
    19	    streak: Math.max(0, Math.min(10000, Math.floor(Number(v.streak) || 0))),
    20	    gamesPlayed: Math.max(0, Math.min(1000000, Math.floor(Number(v.gamesPlayed) || 0))),
    21	    score: Math.max(0, Math.min(100000000, Math.floor(Number(v.score) || 0))),
    22	    lives: Math.max(0, Math.min(3, Math.floor(Number(v.lives ?? 3)))),
    23	    difficulty: difficulties.includes(v.difficulty as WildDifficulty) ? v.difficulty as WildDifficulty : "medium",
    24	    mode: modes.includes(v.mode as WildMode) ? v.mode as WildMode : "normal",
    25	  };
    26	};
    27	
    28	const readSave = (json: string): StoredSave => {
    29	  try { return JSON.parse(json) as StoredSave; } catch { return {}; }
    30	};
    31	
    32	export const loadWildWhispersCloud = createServerFn({ method: "GET" })
    33	  .middleware([authMiddleware])
    34	  .handler(async ({ context }) => {
    35	    const row = await getPrisma().playerSave.findUnique({ where: { userId: context.userId } });
    36	    if (!row) return { ok: true as const, progress: null, revision: 0 };
    37	    const save = readSave(row.saveJson);
    38	    return { ok: true as const, progress: save.wildWhispers ? normalize(save.wildWhispers) : null, revision: Number(row.revision) };
    39	  });

    41	/** Starts an opaque server-authoritative round. The client never submits score/lives/streak. */
    42	export const startWildChallenge = createServerFn({ method: "POST" })
    43	  .middleware([authMiddleware])
    44	  .handler(async ({ context }) => {
    45	    const db = getPrisma();
    46	    return db.$transaction(async (tx) => {
    47	      const row = await tx.playerSave.findUnique({ where: { userId: context.userId } });
    48	      const animal = WILD_ANIMALS[Math.floor(Math.random() * WILD_ANIMALS.length)]!;
    49	      const now = Date.now();
    50	      const challenge = { animalId: animal.id, issuedAt: now };
    51	      if (!row) {
    52	        const save = defaultSave() as StoredSave;
    53	        save.wildWhispers = fallback;
    54	        save.wildWhispersChallenge = challenge;
    55	        await tx.playerSave.create({ data: { userId: context.userId, saveJson: JSON.stringify(save), version: 1, revision: 1n } });
    56	        return { ok: true as const, animalId: animal.id, revision: 1 };
    57	      }
    58	      const save = readSave(row.saveJson);
    59	      save.wildWhispers = normalize(save.wildWhispers ?? fallback);
    60	      save.wildWhispersChallenge = challenge;
    61	      const updated = await tx.playerSave.updateMany({ where: { userId: context.userId, revision: row.revision }, data: { saveJson: JSON.stringify(save), version: { increment: 1 }, revision: { increment: 1n } } });
    62	      if (updated.count !== 1) return { ok: false as const, conflict: true as const };
    63	      return { ok: true as const, animalId: animal.id, revision: Number(row.revision) + 1 };
    64	    });
    65	  });

    67	/** Awards Wild Whispers progress from the server-side challenge only. */
    68	export const answerWildChallenge = createServerFn({ method: "POST" })
    69	  .middleware([authMiddleware])
    70	  .validator((d: { selectedAnimalId: string }) => ({ selectedAnimalId: String(d.selectedAnimalId ?? "").slice(0, 64) }))
    71	  .handler(async ({ context, data }) => {
    72	    const db = getPrisma();
    73	    return db.$transaction(async (tx) => {
    74	      const row = await tx.playerSave.findUnique({ where: { userId: context.userId } });
    75	      if (!row) return { ok: false as const, error: "No active game." };
    76	      const save = readSave(row.saveJson);
    77	      const challenge = save.wildWhispersChallenge;
    78	      if (!challenge || Date.now() - Number(challenge.issuedAt) > 10 * 60 * 1000) return { ok: false as const, error: "Challenge expired. Start a new round." };
    79	      const progress = normalize(save.wildWhispers ?? fallback);
    80	      if (progress.lives <= 0) progress.lives = 3;
    81	      const correct = data.selectedAnimalId === challenge.animalId;
    82	      const nextStreak = correct ? progress.streak + 1 : 0;
    83	      const next: WildProgress = {
    84	        ...progress,
    85	        discovered: correct ? [...new Set([...progress.discovered, challenge.animalId])] : progress.discovered,
    86	        streak: nextStreak,
    87	        bestStreak: Math.max(progress.bestStreak, nextStreak),
    88	        gamesPlayed: Math.min(1000000, progress.gamesPlayed + 1),
    89	        score: Math.min(100000000, progress.score + (correct ? 100 + nextStreak * 10 : 0)),
    90	        lives: correct ? Math.min(3, progress.lives + (nextStreak >= 5 ? 1 : 0)) : Math.max(0, progress.lives - 1),
    91	      };
    92	      save.wildWhispers = next;
    93	      delete save.wildWhispersChallenge;
    94	      const updated = await tx.playerSave.updateMany({ where: { userId: context.userId, revision: row.revision }, data: { saveJson: JSON.stringify(save), version: { increment: 1 }, revision: { increment: 1n } } });
    95	      if (updated.count !== 1) return { ok: false as const, conflict: true as const };
    96	      return { ok: true as const, correct, progress: next, revision: Number(row.revision) + 1, animalId: challenge.animalId };
    97	    });
    98	  });

   100	/** Only settings are client-controlled. Scores, lives, streaks and discoveries are server-controlled. */
   101	export const saveWildWhispersSettings = createServerFn({ method: "POST" })
   102	  .middleware([authMiddleware])
   103	  .validator((d: { mode: string; difficulty: string; expectedRevision?: number }) => ({
   104	    mode: modes.includes(d.mode as WildMode) ? d.mode as WildMode : "medium" as WildMode,
   105	    difficulty: difficulties.includes(d.difficulty as WildDifficulty) ? d.difficulty as WildDifficulty : "medium" as WildDifficulty,
   106	    expectedRevision: Number.isFinite(d.expectedRevision) ? Math.max(0, Math.floor(Number(d.expectedRevision))) : undefined,
   107	  }))
   108	  .handler(async ({ context, data }) => {
   109	    const db = getPrisma();
   110	    return db.$transaction(async (tx) => {
   111	      const row = await tx.playerSave.findUnique({ where: { userId: context.userId } });
   112	      if (!row) return { ok: false as const, error: "Save not initialized." };
   113	      if (data.expectedRevision !== undefined && data.expectedRevision !== Number(row.revision)) return { ok: false as const, conflict: true as const };
   114	      const save = readSave(row.saveJson);
   115	      const progress = normalize(save.wildWhispers ?? fallback);
   116	      save.wildWhispers = { ...progress, mode: data.mode, difficulty: data.difficulty };
   117	      const updated = await tx.playerSave.updateMany({ where: { userId: context.userId, revision: row.revision }, data: { saveJson: JSON.stringify(save), version: { increment: 1 }, revision: { increment: 1n } } });
   118	      if (updated.count !== 1) return { ok: false as const, conflict: true as const };
   119	      return { ok: true as const, progress: save.wildWhispers, revision: Number(row.revision) + 1 };
   120	    });
   121	  });
