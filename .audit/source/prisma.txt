   337	function taggedToSql(strings: TemplateStringsArray, values: unknown[]) {
   338	  let text = strings[0] ?? "";
   339	  const params: unknown[] = [];
   340	  for (let i = 0; i < values.length; i++) {
   341	    const v = values[i];
   342	    params.push(v instanceof Date ? v.toISOString() : typeof v === "bigint" ? Number(v) : v);
   343	    text += `$${params.length}${strings[i + 1] ?? ""}`;
   344	  }
   345	  return { text, params };
   346	}

   362	         JOIN role_permissions rp ON rp.role_id = ur.role_id
   363	         JOIN permissions p ON p.id = rp.permission_id
   364	         WHERE ur.user_id = $1 AND p.name = $2 LIMIT 1`,
   365	        [where.userId, permName],
